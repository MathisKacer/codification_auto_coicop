charger_run <- function(objet_s3) {
  df <- aws.s3::s3read_using(
    FUN = arrow::read_parquet, object = objet_s3, bucket = bucket_s3,
    opts = list(region = "", base_url = endpoint_s3),
  )
  if (!"ligne" %in% names(df)) df$ligne <- seq_len(nrow(df)) - 1L
  df
}

# Troncature niveau 4 AVEC cascade des "0" terminaux -- version validee (cf.
# section 1). Necessaire pour les codes candidats (lcs/rag/ragann/ttc), qui
# n'ont pas d'equivalent "_lvl4" pre-calcule.
tronquer_niveau_un <- function(code, niveau = 4) {
  if (is.na(code)) return(NA_character_)
  s <- as.character(code)
  if (s %in% c("AUCUNE_SUGGESTION", "NON_CODABLE")) return(s)
  n_chiffres_cible <- niveau + 1
  chiffres <- 0L
  out <- character(0)
  for (ch in strsplit(s, "")[[1]]) {
    if (grepl("[0-9]", ch)) {
      if (chiffres == n_chiffres_cible) break
      chiffres <- chiffres + 1L
      out <- c(out, ch)
    } else if (chiffres < n_chiffres_cible) {
      out <- c(out, ch)
    }
  }
  resultat <- paste(out, collapse = "")
  while (grepl("\\.0$", resultat)) resultat <- sub("\\.0$", "", resultat)
  resultat
}
tronquer_niveau <- function(codes, niveau = 4) {
  vapply(codes, tronquer_niveau_un, character(1), niveau = niveau, USE.NAMES = FALSE)
}
tronquer_sans_cascade <- function(codes, niveau = 4) {
  vapply(codes, function(code) {
    if (is.na(code)) return(NA_character_)
    s <- as.character(code)
    if (s %in% c("AUCUNE_SUGGESTION", "NON_CODABLE")) return(s)
    n_chiffres_cible <- niveau + 1
    chiffres <- 0L; out <- character(0)
    for (ch in strsplit(s, "")[[1]]) {
      if (grepl("[0-9]", ch)) {
        if (chiffres == n_chiffres_cible) break
        chiffres <- chiffres + 1L; out <- c(out, ch)
      } else if (chiffres < n_chiffres_cible) out <- c(out, ch)
    }
    paste(out, collapse = "")
  }, character(1), USE.NAMES = FALSE)
}

construire_table_long <- function(df, niveau = 4, fn_tronquer = tronquer_niveau) {
  # vrai_n4 doit deja exister dans df (code_lvl4 si dispo, sinon fn_tronquer(code)).
  # fn_tronquer parametrable pour pouvoir rejouer le pipeline SANS le fix
  # cascade (tronquer_sans_cascade), a des fins de comparaison (section 6).
  df <- df %>%
    mutate(
      lcs_n4    = fn_tronquer(lcs_code, niveau),
      rag_n4    = fn_tronquer(rag_code, niveau),
      ragann_n4 = fn_tronquer(ragann_code, niveau),
      ttc_n4    = fn_tronquer(ttc_code_1, niveau),
    )
  candidats <- df %>%
    select(ligne, lcs_n4, rag_n4, ragann_n4, ttc_n4, vrai_n4) %>%
    pivot_longer(
      cols = c(lcs_n4, rag_n4, ragann_n4, ttc_n4),
      names_to = "classifieur", values_to = "code_candidat",
    ) %>%
    filter(!is.na(code_candidat)) %>%
    distinct(ligne, code_candidat, vrai_n4)

  candidats %>%
    left_join(
      df %>% select(
        ligne, lcs_n4, rag_n4, ragann_n4, ttc_n4,
        lcs_distance, rag_confidence, ragann_confidence, ttc_conf_1,
        budget, shop_type_name, source,
      ),
      by = "ligne",
    ) %>%
    mutate(
      vote_lcs    = as.integer(!is.na(lcs_n4)    & lcs_n4    == code_candidat),
      vote_rag    = as.integer(!is.na(rag_n4)    & rag_n4    == code_candidat),
      vote_ragann = as.integer(!is.na(ragann_n4) & ragann_n4 == code_candidat),
      vote_ttc    = as.integer(!is.na(ttc_n4)    & ttc_n4    == code_candidat),
      nb_votants  = vote_lcs + vote_rag + vote_ragann + vote_ttc,
      correcte    = as.integer(!is.na(vrai_n4) & code_candidat == vrai_n4),
      code_candidat_n1 = fn_tronquer(code_candidat, niveau = 1),
      conf_rag    = if_else(vote_rag == 1L,    coalesce(rag_confidence, -1),    -1),
      conf_ragann = if_else(vote_ragann == 1L, coalesce(ragann_confidence, -1), -1),
      conf_ttc    = if_else(vote_ttc == 1L,    coalesce(ttc_conf_1, -1),        -1),
      dist_lcs    = if_else(vote_lcs == 1L,    coalesce(lcs_distance, 1.5),     1.5),
      shop_type_name = if_else(is.na(shop_type_name), "INCONNU", shop_type_name),
      source         = if_else(is.na(source), "INCONNU", source),
    ) %>%
    select(
      ligne, code_candidat, code_candidat_n1,
      vote_lcs, vote_rag, vote_ragann, vote_ttc, nb_votants,
      conf_rag, conf_ragann, conf_ttc, dist_lcs, correcte,
      budget, shop_type_name, source,
    )
}

1. Troncature niveau 4 : le fix cascade des “0” terminaux

Ce fix (retrait en cascade des “0” terminaux dans tronquer_niveau()) n’est pas spécifique à SIRUS : c’est la même logique que src/coicop.py::tronquer_niveau(), utilisée par toutes les pages de ce site. Sa légitimité structurelle (aucune position, sur les 3 runs, où “0” entre en concurrence avec un vrai chiffre) est démontrée une fois pour toutes dans Troncature niveau 4 plutôt que répétée ici.

Ce qui est spécifique à ce document : depuis le run codif-x98xl (2026-07-30), une colonne code_lvl4 est calculée en amont du pipeline pour le vrai code. Vérification qu’elle correspond bien au fix cascade (nécessaire puisque code_lvl4, pas code brut, sert de vérité terrain dans tout le reste de ce document, section 4 comprise) :

2b. Sentinelles de confiance quand un classifieur n’a pas voté

Quand un classifieur ne vote pas pour un candidat donné, sa confiance est mise à une sentinelle hors de la plage réelle des valeurs (-1 pour les confiances, bornées [0, 1] en pratique ; 1.5 pour lcs_distance, bornée [0, 0.8]) plutôt qu’à 0 ou NA. Vérifié non problématique pour un modèle à seuils comme SIRUS :

  • Sentinelle toujours hors plage réelle (pas de collision avec une vraie valeur).
  • Redondant avec vote_lcs/rag/ragann/ttc (0/1) : aucune perte d’information, et permet à SIRUS de fusionner en une seule règle “vote faible OU absent” (ex. if conf_ttc < 0.78), ce qui est le sens voulu.
  • Cas limite vérifié sur données réelles : 0 occurrence où un classifieur a voté (code non manquant) mais avec une confiance NA – le coalesce() de secours ne se déclenche jamais actuellement.

diagnostiquer_run <- function(df, vrai_n4, nom_run) {
  bind_rows(lapply(c("lcs_code", "rag_code", "ragann_code", "ttc_code_1"), function(col) {
    cand_n4 <- tronquer_niveau(df[[col]], niveau)
    taux <- mean(!is.na(cand_n4) & !cand_n4 %in% c("AUCUNE_SUGGESTION", "NON_CODABLE") & cand_n4 == vrai_n4, na.rm = TRUE)
    data.frame(run = nom_run, classifieur = col, taux_match = taux)
  }))
}

7. Système à deux étages : SIRUS confiant → RF sur les incertains → reprise manuelle

Exploration menée dans test.qmd sur le run codif-8m8jn (premier nouveau run), non répétée depuis. Principe : sous le seuil proba_gagnant = 0.6 (zone où SIRUS abandonne), un Random Forest entraîné spécifiquement sur ce sous-ensemble incertain rattrape une partie des cas (rappel 59.7 % contre 17.5 % pour SIRUS seul sur les mêmes lignes de test). Bilan du système à 2 étages, tel qu’obtenu dans test.qmd :

Étape % volume total Fiabilité
SIRUS confiant (auto) ~63 % ~98 %
RF confiant sur incertains (auto) variable selon seuil_rf variable
Reprise manuelle / LLM le reste —

Comparé à SIRUS seul au même seuil (80.1 % de fiabilité globale, 37.1 % de volume en reprise), la cascade à 2 étages réduit le volume de reprise nécessaire pour une fiabilité globale comparable ou meilleure – à confirmer sur un échantillon plus grand (le RF n’a été validé que sur 20 % de la zone incertaine). Non recalculé sur x98xl : la zone incertaine et les performances du RF dépendraient de la régression RAG-ANN identifiée section 4, à ré-évaluer si cette piste est reprise.

Rappel du message le plus important de cette exploration : ce système optimise la décision en aval, avec les classifieurs de base tels qu’ils sont. Il ne corrige pas la régression RAG-ANN elle-même – aucun seuil, aucun second modèle ne peut récupérer les produits pour lesquels le bon code n’est proposé par aucun classifieur.

8. Synthèse et recommandations

  • Le fix de troncature “.0” est légitime (section 1), validé structurellement sur l’ensemble des codes observés (0 position à risque sur 3 runs) – et maintenant remplacé en amont par code_lvl4 sur les runs qui l’incluent.
  • Le modèle SIRUS de base (10 règles, profondeur 2) est déjà quasi-optimal (section 5) : ni plus de règles, ni de features additionnelles (budget/shop_type_name/source, section 3) n’apportent de gain mesurable.
  • Le vrai facteur limitant est en amont : la régression RAG-ANN, présente depuis le 23 juillet et toujours non corrigée au 30 juillet (section 4), plafonne la borne haute à ~88 % contre ~98 % sur l’ancien run. Aucun réglage de SIRUS ne peut compenser un candidat qui n’est simplement pas proposé.
  • Prochaine étape recommandée : corriger RAG-ANN en amont plutôt que continuer à optimiser le re-ranking. En parallèle, le système à 2 étages (section 7) reste une option pour réduire le volume de reprise manuelle en attendant, à re-valider sur x98xl si retenu.
